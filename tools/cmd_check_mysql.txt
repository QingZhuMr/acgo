#!/bin/bash
# 检查 MySQL 服务状态和 root 密码
echo "---MySQL服务状态---"
systemctl status mysql 2>&1 | head -5
echo "---尝试无密码 socket---"
mysql -uroot -S /tmp/mysql.sock -e "SELECT 1" 2>&1
echo "---检查MySQL相关配置---"
cat /root/.my.cnf 2>&1
echo "---检查mysqladmin---"
/www/server/mysql/bin/mysqladmin -uroot status 2>&1 | head -3
